using Microsoft.AspNetCore.Mvc;

using PCS.API.Security;

using PCS.Application.Common.Security;
using PCS.Application.Planning.Interfaces;

namespace PCS.API.Controllers;

[ApiController]
[Route("api/pms")]
public class PmsController : ControllerBase
{
    private static readonly string[] AllowedExtensions =
    {
        ".xlsx",
        ".xlsm",
        ".xltx",
        ".xltm"
    };

    private readonly IPmsExcelService _excelService;
    private readonly IPmsService _pmsService;
    private readonly IPmsTemplateService _templateService;

    public PmsController(
        IPmsExcelService excelService,
        IPmsService pmsService,
        IPmsTemplateService templateService)
    {
        _excelService = excelService;
        _pmsService = pmsService;
        _templateService = templateService;
    }

    [HttpPost("import")]
    [Permission(Permissions.ProjectEdit)]
    public async Task<IActionResult> Import(
        IFormFile file,
        [FromQuery] Guid projectId)
    {
        if (projectId == Guid.Empty)
        {
            return BadRequest(
                "ProjectId is required.");
        }

        if (file == null || file.Length == 0)
        {
            return BadRequest(
                "Excel file is empty.");
        }

        var extension =
            Path.GetExtension(file.FileName)
                .ToLowerInvariant();

        if (!AllowedExtensions.Contains(extension))
        {
            return BadRequest(
                $"Extension '{extension}' is not supported.");
        }

        var tempPath =
            Path.Combine(
                Path.GetTempPath(),
                $"{Guid.NewGuid()}{extension}");

        try
        {
            await using (var stream =
                System.IO.File.Create(tempPath))
            {
                await file.CopyToAsync(stream);
            }

            var dto =
                await _excelService.ReadPmsExcelAsync(
                    tempPath);

            dto.ProjectId = projectId;

            await _pmsService.ImportPmsAsync(dto);

            return Ok(new
            {
                Success = true,
                Message =
                    "PMS imported successfully.",
                FileName =
                    file.FileName
            });
        }
        catch (Exception ex)
        {
            return BadRequest(new
            {
                Success = false,
                Message =
                    ex.Message,
                Inner =
                    ex.InnerException?.Message
            });
        }
        finally
        {
            if (System.IO.File.Exists(tempPath))
            {
                System.IO.File.Delete(tempPath);
            }
        }
    }

    [HttpGet("export/{projectId}")]
    [Permission(Permissions.ProjectView)]
    public async Task<IActionResult> Export(
        Guid projectId)
    {
        if (projectId == Guid.Empty)
        {
            return BadRequest(
                "ProjectId is required.");
        }

        var dto =
            await _pmsService.ExportPmsAsync(
                projectId);

        var file =
            await _excelService.GeneratePmsExcelAsync(
                dto);

        return File(
            file,
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
            "PCS-PMS-Report.xlsx");
    }

    [HttpGet("{projectId}/activities")]
    [Permission(Permissions.ProjectView)]
    public async Task<IActionResult> GetActivities(
        Guid projectId)
    {
        if (projectId == Guid.Empty)
        {
            return BadRequest(
                "ProjectId is required.");
        }

        var result =
            await _pmsService.GetActivitiesAsync(
                projectId);

        return Ok(result);
    }

    [HttpGet("{projectId}/wbs")]
    [Permission(Permissions.ProjectView)]
    public async Task<IActionResult> GetWbs(
        Guid projectId)
    {
        if (projectId == Guid.Empty)
        {
            return BadRequest(
                "ProjectId is required.");
        }

        var result =
            await _pmsService.GetWbsTreeAsync(
                projectId);

        return Ok(result);
    }

    [HttpGet("template")]
    [Permission(Permissions.ProjectView)]
    public IActionResult DownloadTemplate()
    {
        var file =
            _templateService.GenerateTemplate();

        return File(
            file,
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
            "PCS-PMS-Import-Template.xlsx");
    }
}

namespace PCS.Application.Planning.Services;

public class PmsService : IPmsService
{
    private readonly IGenericRepository<Project> _projectRepository;
    private readonly IGenericRepository<Company> _companyRepository;
    private readonly IGenericRepository<WbsNode> _wbsRepository;
    private readonly IGenericRepository<Activity> _activityRepository;
    private readonly IUnitOfWork _unitOfWork;
    private readonly IWbsCalculationService _wbsCalculationService;


    public PmsService(
        IGenericRepository<Project> projectRepository,
        IGenericRepository<Company> companyRepository,
        IGenericRepository<WbsNode> wbsRepository,
        IGenericRepository<Activity> activityRepository,
        IUnitOfWork unitOfWork,
        IWbsCalculationService wbsCalculationService)
    {
        _projectRepository = projectRepository;
        _companyRepository = companyRepository;
        _wbsRepository = wbsRepository;
        _activityRepository = activityRepository;
        _unitOfWork = unitOfWork;
        _wbsCalculationService = wbsCalculationService;

    // =====================================================
    // IMPORT PMS
    // =====================================================

    public async Task ImportPmsAsync(
        ImportPmsDto dto)
    {
        if(dto.ProjectId == Guid.Empty)
            throw new Exception(
                "ProjectId is required.");



        var project =
            await _projectRepository
                .GetByIdAsync(dto.ProjectId);



        if(project == null)
        {
            throw new Exception(
                $"Project '{dto.ProjectId}' does not exist.");
        }




        var existingWbs =
            (await _wbsRepository.GetAllAsync())
            .Where(x =>
                x.ProjectId == dto.ProjectId)
            .ToDictionary(

    // =====================================================
    // BUILD WBS TREE
    // =====================================================

    private static List<WbsNodeDto> BuildWbsTree(
        List<WbsNode> nodes,
        List<Activity> activities,
        decimal totalProjectBudget)
    {

        var budgetLookup =
            activities
            .GroupBy(
                x => x.WbsNodeId)
            .ToDictionary(
                x => x.Key,
                x => x.Sum(
                    a => a.Budget));



        var lookup =
            nodes.ToDictionary(
                x => x.Id,
                x =>
                new WbsNodeDto
                {
                    Id =
                        x.Id,


                    ParentId =
                        x.ParentId,


                    Code =
                        x.Code,


                    ParentCode =
                        nodes
                        .FirstOrDefault(
                            p =>
                                p.Id ==
                                x.ParentId)
                        ?.Code,


                    Path =
                        x.Path,


                    Name =
                        x.Name,


                    Level =
                        x.Level,


                    SortOrder =
                        x.SortOrder
                });



        // ==========================================
        // Bottom-Up Budget Calculation
        // ==========================================

        foreach(var node in nodes
            .OrderByDescending(
                x => x.Level))
        {

            var dto =
                lookup[node.Id];


            decimal budget = 0;



            if(budgetLookup.TryGetValue(
                    node.Id,
                    out var activityBudget))
            {
                budget += activityBudget;
            }



            budget +=
                dto.Children
                .Sum(
                    x => x.Budget);



            dto.Budget =
                budget;
        }





        // ==========================================
        // W.V Calculation
        // ==========================================

        foreach(var item in lookup.Values)
        {
            item.WeightValue =
                totalProjectBudget == 0
                ? 0
                :
                (item.Budget /
                 totalProjectBudget)
                 * 100m;
        }





        // ==========================================
        // W.F Calculation
        // ==========================================

        foreach(var item in lookup.Values)
        {
            if(item.ParentId.HasValue &&
               lookup.TryGetValue(
                    item.ParentId.Value,
                    out var parent))
            {
                item.WeightFactor =
                    parent.Budget == 0
                    ? 0
                    :
                    (item.Budget /
                     parent.Budget)
                     * 100m;
            }
            else
            {
                item.WeightFactor =
                    item.WeightValue;
            }
        }





        var roots =
            new List<WbsNodeDto>();


        foreach(var node in nodes)
        {
            var item =
                lookup[node.Id];


            if(node.ParentId.HasValue &&
               lookup.TryGetValue(
                    node.ParentId.Value,
                    out var parent))
            {
                parent.Children.Add(item);
            }
            else
            {
                roots.Add(item);
            }
        }


        return roots;
    }

using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;


using PCS.Application.Common.Interfaces;
using PCS.Application.Identity.Interfaces;
using PCS.Application.Projects.Interfaces;
using PCS.Application.Planning.Interfaces;
using PCS.Application.Licensing.Interfaces;

using PCS.Infrastructure.BackgroundJobs;
using PCS.Infrastructure.Persistence;
using PCS.Infrastructure.Repositories;
using PCS.Infrastructure.Services;
using PCS.Infrastructure.Services.Identity;

namespace PCS.Infrastructure;

public static class DependencyInjection
{
    public static IServiceCollection AddInfrastructure(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        // ==========================
        // Database
        // ==========================

        string? connectionString =
            configuration.GetConnectionString(
                "DefaultConnection");

        if (string.IsNullOrWhiteSpace(connectionString))
        {
            throw new InvalidOperationException(
                "PCS database connection string was not configured.");
        }

        services.AddDbContext<PCSDbContext>(
            options =>
            {
                if (IsSqlServerConnectionString(connectionString))
                {
                    options.UseSqlServer(connectionString);
                }
                else
                {
                    options.UseSqlite(connectionString);
                }
            });


        // ==========================
        // Repositories
        // ==========================

        services.AddScoped(
            typeof(IGenericRepository<>),
            typeof(GenericRepository<>));


        services.AddScoped<
            IProjectRepository,
            ProjectRepository>();


        services.AddScoped<
            IActivityRepository,
            ActivityRepository>();


        services.AddScoped<
            IActivityRelationshipRepository,
            ActivityRelationshipRepository>();


        // ==========================
        // Persistence
        // ==========================

        services.AddScoped<
            IUnitOfWork,
            UnitOfWork>();


        // ==========================
        // Identity / Security
        // ==========================

        services.AddScoped<
         IPasswordHasher,
         PasswordHasher>();


        services.AddScoped<
         IJwtService,
         JwtService>();


        services.AddScoped<
         IAuthService,
         AuthService>();


        // ==========================
        // Services
        // ==========================

        services.AddScoped<
            IAuditService,
            AuditService>();


        // ==========================
        // Licensing
        // ==========================

        services.AddScoped<
            ILicenseService,
            LicenseService>();


        // ==========================
        // Background Jobs
        // ==========================

        services.AddScoped<
            IBackgroundJobService,
            HangfireBackgroundJobService>();


        return services;
    }


    private static bool IsSqlServerConnectionString(
        string connectionString)
    {
        string value =
            connectionString.Trim();


        // ========================================
        // SQL Server connection string indicators
        // ========================================

        if (value.Contains(
                "Server=",
                StringComparison.OrdinalIgnoreCase))
        {
            return true;
        }


        if (value.Contains(
                "Data Source=",
                StringComparison.OrdinalIgnoreCase) &&
            !value.Contains(
                ".db",
                StringComparison.OrdinalIgnoreCase) &&
            !value.Contains(
                ".sqlite",
                StringComparison.OrdinalIgnoreCase))
        {
            return true;
        }


        if (value.Contains(
                "Initial Catalog=",
                StringComparison.OrdinalIgnoreCase))
        {
            return true;
        }


        if (value.Contains(
                "Trusted_Connection=",
                StringComparison.OrdinalIgnoreCase))
        {
            return true;
        }


        if (value.Contains(
                "Integrated Security=",
                StringComparison.OrdinalIgnoreCase))
        {
            return true;
        }


        // ========================================
        // Otherwise treat as SQLite
        // ========================================

        return false;
    }
}

function clamp(
    value: number,
    min: number,
    max: number
): number {

    return Math.min(
        Math.max(value, min),
        max
    );
}


/* ============================================================
   FLATTEN WBS TREE
   ============================================================ */

function flattenWbsNodes(
    nodes: WbsNode[]
): WbsNode[] {

    const result: WbsNode[] = [];

    const visit = (
        items: WbsNode[]
    ) => {

        const sorted =
            [...items].sort(
                (a, b) =>
                    (a.sortOrder ?? 0) -
                    (b.sortOrder ?? 0)
            );

        for (const node of sorted) {

            result.push(node);

            if (
                Array.isArray(node.children) &&
                node.children.length > 0
            ) {

                visit(
                    node.children
                );
            }
        }
    };

    visit(nodes);

    return result;
}


/* ============================================================
   FLATTEN ACTIVITIES
   ============================================================ */

function flattenActivities(
    rows: GanttRow[]
): GanttRow[] {

    const result: GanttRow[] = [];

    for (const row of rows) {

        if (
            row.type === "activity"
        ) {

            result.push(row);

            continue;
        }

        if (
            row.children.length > 0
        ) {

            result.push(
                ...flattenActivities(
                    row.children
                )
            );
        }
    }

    return result;
}

function buildHierarchicalRows(
    wbsTree: WbsNode[],
    activities: GanttActivity[]
): GanttRow[] {

    const allWbsNodes =
        flattenWbsNodes(
            wbsTree
        );


    /* ========================================================
       ACTIVITY ROWS
       ======================================================== */

    const activityRows =
        new Map<string, GanttRow>();


    for (const activity of activities) {

        const start =
            Math.max(
                0,
                Number(
                    activity.start
                ) || 0
            );


        const duration =
            Math.max(
                0,
                Number(
                    activity.duration
                ) || 0
            );


        const finish =
            Math.max(
                Number(
                    activity.finish
                ) || 0,
                start + duration
            );


        const activityRow: GanttRow = {

            id:
                activity.id,

            type:
                "activity",

            parentId:
                activity.wbsNodeId ?? null,

            code:
                activity.code,

            name:
                activity.name,

            level:
                Math.max(
                    0,
                    Number(
                        activity.wbsLevel ?? 0
                    )
                ) + 1,


            /* ==================================================
               RELATIVE CPM
               ================================================== */

            start,

            finish,

            duration,


            /* ==================================================
               REAL SCHEDULE DATES
               ================================================== */

            plannedStart:
                activity.plannedStart ?? null,

            plannedFinish:
                activity.plannedFinish ?? null,

            actualStart:
                activity.actualStart ?? null,

            actualFinish:
                activity.actualFinish ?? null,

            remainingDuration:
                Math.max(
                    0,
                    Number(
                        activity.remainingDuration
                    ) || 0
                ),


            /* ==================================================
               SCHEDULE METADATA
               ================================================== */

            description:
                activity.description ?? null,

            constraintType:
                activity.constraintType ?? "ASAP",

            constraintDate:
                activity.constraintDate ?? null,

            status:
                activity.status ?? "NotStarted",


            /* ==================================================
               PROGRESS
               ================================================== */

            progress:
                clamp(
                    Number(
                        activity.progress
                    ) || 0,
                    0,
                    100
                ),


            /* ==================================================
               FLOAT
               ================================================== */

            float:
                Math.max(
                    0,
                    Number(
                        activity.float
                    ) || 0
                ),

            totalFloat:
                Math.max(
                    0,
                    Number(
                        activity.totalFloat
                    ) || 0
                ),

            freeFloat:
                Math.max(
                    0,
                    Number(
                        activity.freeFloat
                    ) || 0
                ),

            drivingFloat:
                Math.max(
                    0,
                    Number(
                        activity.drivingFloat
                    ) || 0
                ),


            /* ==================================================
               STATUS
               ================================================== */

            isCritical:
                Boolean(
                    activity.isCritical
                ),

            isMilestone:
                Boolean(
                    activity.isMilestone
                ),


            /* ==================================================
               RELATIONSHIPS
               ================================================== */

            predecessors:
                activity.predecessors ?? [],

            successors:
                activity.successors ?? [],


            children: []
        };


        activityRows.set(
            activity.id,
            activityRow
        );
    }


    /* ========================================================
       WBS ROWS
       ======================================================== */

    const wbsRows =
        new Map<string, GanttRow>();


    for (const node of allWbsNodes) {

        const row: GanttRow = {

            id:
                node.id,

            type:
                "wbs",

            parentId:
                node.parentId ?? null,

            code:
                node.code,

            name:
                node.name,

            level:
                Math.max(
                    0,
                    Number(
                        node.level
                    ) || 0
                ),


            start: 0,

            finish: 0,

            duration: 0,


            /*
             * PMS WBS progress is authoritative.
             */

            progress:
                clamp(
                    Number(
                        node.progress
                    ) || 0,
                    0,
                    100
                ),


            float: 0,

            totalFloat: 0,

            freeFloat: 0,

            drivingFloat: 0,

            isCritical: false,

            isMilestone: false,

            children: []
        };


        wbsRows.set(
            node.id,
            row
        );
    }


    /* ========================================================
       ATTACH ACTIVITIES TO WBS
       ======================================================== */

    for (
        const activity of activities
    ) {

        const wbsId =
            activity.wbsNodeId;


        if (!wbsId) {
            continue;
        }


        const wbsRow =
            wbsRows.get(
                wbsId
            );


        const activityRow =
            activityRows.get(
                activity.id
            );


        if (
            !wbsRow ||
            !activityRow
        ) {
            continue;
        }


        wbsRow.children.push(
            activityRow
        );
    }


    /* ========================================================
       ATTACH WBS TO PARENTS
       ======================================================== */

    for (
        const node of allWbsNodes
    ) {

        const currentRow =
            wbsRows.get(
                node.id
            );


        if (!currentRow) {
            continue;
        }


        if (node.parentId) {

            const parentRow =
                wbsRows.get(
                    node.parentId
                );


            if (parentRow) {

                parentRow.children.push(
                    currentRow
                );
            }
        }
    }


    /* ========================================================
       SORT TREE
       ======================================================== */

    const sortTree = (
        rows: GanttRow[]
    ): GanttRow[] => {

        rows.sort(
            (a, b) => {

                if (
                    a.type !== b.type
                ) {

                    return a.type === "wbs"
                        ? -1
                        : 1;
                }


                if (
                    a.type === "wbs" &&
                    b.type === "wbs"
                ) {

                    const aNode =
                        allWbsNodes.find(
                            x =>
                                x.id === a.id
                        );


                    const bNode =
                        allWbsNodes.find(
                            x =>
                                x.id === b.id
                        );


                    const sortDifference =
                        (
                            aNode?.sortOrder ?? 0
                        ) -
                        (
                            bNode?.sortOrder ?? 0
                        );


                    if (
                        sortDifference !== 0
                    ) {

                        return sortDifference;
                    }


                    return a.code.localeCompare(
                        b.code,
                        undefined,
                        {
                            numeric: true,
                            sensitivity: "base"
                        }
                    );
                }


                return a.code.localeCompare(
                    b.code,
                    undefined,
                    {
                        numeric: true,
                        sensitivity: "base"
                    }
                );
            }
        );


        for (
            const row of rows
        ) {

            if (
                row.children.length > 0
            ) {

                sortTree(
                    row.children
                );
            }
        }


        return rows;
    };


    /* ========================================================
       CALCULATE SUMMARY
       ======================================================== */

    const calculateSummary = (
        row: GanttRow
    ): {
        start: number;
        finish: number;
        critical: boolean;
    } => {

        if (
            row.type === "activity"
        ) {

            return {

                start:
                    row.start,

                finish:
                    row.finish,

                critical:
                    row.isCritical
            };
        }


        if (
            row.children.length === 0
        ) {

            row.start = 0;
            row.finish = 0;
            row.duration = 0;
            row.isCritical = false;

            return {

                start: 0,

                finish: 0,

                critical: false
            };
        }


        const childResults =
            row.children.map(
                child =>
                    calculateSummary(
                        child
                    )
            );


        const validChildren =
            childResults.filter(
                child =>
                    Number.isFinite(
                        child.start
                    ) &&
                    Number.isFinite(
                        child.finish
                    ) &&
                    (
                        child.finish > 0 ||
                        child.start > 0
                    )
            );


        if (
            validChildren.length === 0
        ) {

            row.start = 0;
            row.finish = 0;
            row.duration = 0;

            row.isCritical =
                childResults.some(
                    child =>
                        child.critical
                );

            return {

                start: 0,

                finish: 0,

                critical:
                    row.isCritical
            };
        }


        row.start =
            Math.min(
                ...validChildren.map(
                    child =>
                        child.start
                )
            );


        row.finish =
            Math.max(
                ...validChildren.map(
                    child =>
                        child.finish
                )
            );


        row.duration =
            Math.max(
                0,
                row.finish -
                row.start
            );


        row.isCritical =
            childResults.some(
                child =>
                    child.critical
            );


        return {

            start:
                row.start,

            finish:
                row.finish,

            critical:
                row.isCritical
        };
    };


    /* ========================================================
       ROOT ROWS
       ======================================================== */

    const roots =
        Array.from(
            wbsRows.values()
        ).filter(
            row =>
                !row.parentId
        );


    /* ========================================================
       ORPHAN ACTIVITIES
       ======================================================== */

    const orphanActivities =
        Array.from(
            activityRows.values()
        ).filter(
            activity =>
                !activity.parentId ||
                !wbsRows.has(
                    activity.parentId
                )
        );


    roots.push(
        ...orphanActivities
    );


    /* ========================================================
       FINAL SORT
       ======================================================== */

    const sortedRoots =
        sortTree(
            roots
        );


    for (
        const root of sortedRoots
    ) {

        calculateSummary(
            root
        );
    }


    return sortedRoots;
}

export default function ProjectSchedule() {

    const { id } =
        useParams<{
            id: string
        }>();


    const [
        data,
        setData
    ] =
        useState<GanttProject | null>(
            null
        );


    const [
        wbs,
        setWbs
    ] =
        useState<WbsNode[]>([]);


    const [
        error,
        setError
    ] =
        useState<string | null>(
            null
        );


    const [
        loading,
        setLoading
    ] =
        useState(true);


    /* ========================================================
       LOAD DATA
       ======================================================== */

    useEffect(() => {

        let cancelled = false;


        async function load() {

            if (!id) {

                setError(
                    "Project ID not found"
                );

                setLoading(false);

                return;
            }


            setLoading(true);
            setError(null);


            try {

                const [
                    ganttResponse,
                    wbsResponse
                ] =
                    await Promise.all([
                        getProjectGantt(id),
                        getPmsWbs(id)
                    ]);


                if (
                    cancelled
                ) {
                    return;
                }


                if (
                    !ganttResponse ||
                    !Array.isArray(
                        ganttResponse.activities
                    )
                ) {

                    throw new Error(
                        "Invalid schedule data received"
                    );
                }


                setData(
                    ganttResponse
                );


                setWbs(
                    Array.isArray(
                        wbsResponse
                    )
                        ? wbsResponse
                        : []
                );

            }
            catch (err) {

                if (
                    cancelled
                ) {
                    return;
                }


                console.error(
                    "PROJECT SCHEDULE ERROR:",
                    err
                );


                setError(
                    "Failed to load project schedule"
                );

            }
            finally {

                if (
                    !cancelled
                ) {

                    setLoading(false);
                }
            }
        }


        load();


        return () => {

            cancelled = true;
        };

    }, [id]);


    /* ========================================================
       BUILD HIERARCHICAL DATA
       ======================================================== */

    const hierarchicalData =
        useMemo<
            HierarchicalGanttProject | null
        >(() => {

            if (!data) {
                return null;
            }


            const rows =
                buildHierarchicalRows(

                    data.activities
                );


            const flattenedActivities =
                flattenActivities(
                    rows
                );


            const flattenedWbs =
                flattenWbsNodes(
                    wbs
                );


            const criticalCount =
                flattenedActivities.filter(
                    row =>
                        row.isCritical
                ).length;


            const milestoneCount =
                flattenedActivities.filter(
                    row =>
                        row.isMilestone
                ).length;


            return {

                projectId:
                    data.projectId,

                duration:
                    data.duration,

                rows,

                activityCount:
                    flattenedActivities.length,

                wbsCount:
                    flattenedWbs.length,

                criticalCount,

                milestoneCount
            };

        }, [
            data,
            wbs
        ]);


    /* ========================================================
       LOADING
       ======================================================== */

    if (loading) {

        return (
            <Box
                p={5}
                display="flex"
                alignItems="center"
                gap={2}
            >

                <CircularProgress
                    size={22}
                />

                <Typography
                    color="text.secondary"
                >
                    Loading Project Schedule...
                </Typography>

            </Box>
        );
    }


    /* ========================================================
       ERROR
       ======================================================== */

    if (error) {

        return (
            <Box p={3}>

                <Alert severity="error">
                    {error}
                </Alert>

            </Box>
        );
    }


    if (
        !data ||
        !hierarchicalData
    ) {

        return null;
    }


    /* ========================================================
       RENDER
       ======================================================== */

    return (
        <Box
            sx={{
                width: "100%",
                minWidth: 0
            }}
        >

            {/* =================================================
                PAGE HEADER
            ================================================= */}

            <Box

            ================================================= */}

            <Box
                sx={{
                    mt: 0

                    criticalCount={

                    }
                    milestoneCount={

                    }
                />

    );
}

    const childCount = (row.children ?? []).reduce(
        (sum, child) => sum + countWbsRows(child),
        0
    );
    return (row.type === "wbs" && row.level > 0 ? 1 : 0) + childCount;
}

import { useCallback, useEffect, useMemo, useState } from "react";
import type { GanttRow } from "../../../api/ganttApi";

export type GanttExpansionLevel = number | "all";

interface UseGanttTreeResult {
    visibleRows: GanttRow[];
    expandedIds: Set<string>;
    allWbsIds: string[];

    expansionLevel: GanttExpansionLevel;
    maxWbsLevel: number;

    toggleRow: (rowId: string) => void;
    expandAll: () => void;
    collapseAll: () => void;
    expandToLevel: (level: number) => void;
}

function collectWbsIds(
    rows: GanttRow[],
    result: string[] = []
): string[] {
    for (const row of rows) {
        if (row.type === "wbs") {
            result.push(row.id);
        }

        if (row.children?.length) {
            collectWbsIds(row.children, result);
        }
    }

    return result;
}

function collectWbsIdsToLevel(
    rows: GanttRow[],
    maxExpandedLevel: number,
    result: string[] = []
): string[] {
    for (const row of rows) {
        if (
            row.type === "wbs" &&
            row.children?.length &&
            row.level < maxExpandedLevel
        ) {
            result.push(row.id);
        }

        if (row.children?.length) {
            collectWbsIdsToLevel(
                row.children,
                maxExpandedLevel,
                result
            );
        }
    }

    return result;
}

function getMaxWbsLevel(
    rows: GanttRow[],
    current = 0
): number {
    let maxLevel = current;

    for (const row of rows) {
        if (row.type === "wbs") {
            maxLevel = Math.max(
                maxLevel,
                row.level
            );
        }

        if (row.children?.length) {
            maxLevel = Math.max(
                maxLevel,
                getMaxWbsLevel(
                    row.children,
                    maxLevel
                )
            );
        }
    }

    return maxLevel;
}

function flattenVisibleRows(
    rows: GanttRow[],
    expandedIds: Set<string>,
    result: GanttRow[] = []
): GanttRow[] {
    for (const row of rows) {
        result.push(row);

        if (
            row.type === "wbs" &&
            row.children?.length &&
            expandedIds.has(row.id)
        ) {
            flattenVisibleRows(
                row.children,
                expandedIds,
                result
            );
        }
    }

    return result;
}

export function useGanttTree(
    rows: GanttRow[]
): UseGanttTreeResult {
    const allWbsIds = useMemo(
        () => collectWbsIds(rows),
        [rows]
    );

    const maxWbsLevel = useMemo(
        () => getMaxWbsLevel(rows),
        [rows]
    );

    const [
        expandedIds,
        setExpandedIds
    ] = useState<Set<string>>(
        () => new Set(allWbsIds)
    );

    const [
        expansionLevel,
        setExpansionLevel
    ] = useState<GanttExpansionLevel>(
        "all"
    );

    useEffect(() => {
        setExpandedIds(previous => {
            const validIds =
                new Set(allWbsIds);

            const next =
                new Set<string>();

            for (const id of previous) {
                if (validIds.has(id)) {
                    next.add(id);
                }
            }

            if (
                next.size === previous.size &&
                [...next].every(id =>
                    previous.has(id)
                )
            ) {
                return previous;
            }

            return next;
        });

        if (allWbsIds.length === 0) {
            setExpansionLevel("all");
        }
    }, [allWbsIds]);

    const toggleRow =
        useCallback(
            (rowId: string) => {
                setExpandedIds(previous => {
                    const next =
                        new Set(previous);

                    if (
                        next.has(rowId)
                    ) {
                        next.delete(rowId);
                    } else {
                        next.add(rowId);
                    }

                    return next;
                });

                setExpansionLevel("all");
            },
            []
        );

    const expandAll =
        useCallback(() => {
            setExpandedIds(
                new Set(allWbsIds)
            );

            setExpansionLevel("all");
        }, [allWbsIds]);

    const collapseAll =
        useCallback(() => {
            setExpandedIds(
                new Set()
            );

            setExpansionLevel(0);
        }, []);

    const expandToLevel =
        useCallback(
            (level: number) => {
                const normalizedLevel =
                    Math.max(
                        0,
                        Math.min(
                            Math.floor(level),
                            maxWbsLevel
                        )
                    );

                const nextExpandedIds =
                    normalizedLevel <= 0
                        ? new Set<string>()
                        : new Set(
                              collectWbsIdsToLevel(
                                  rows,
                                  normalizedLevel
                              )
                          );

                setExpandedIds(
                    nextExpandedIds
                );

                setExpansionLevel(
                    normalizedLevel
                );
            },
            [
                rows,
                maxWbsLevel
            ]
        );

    const visibleRows =
        useMemo(
            () =>
                flattenVisibleRows(
                    rows,
                    expandedIds
                ),
            [
                rows,
                expandedIds
            ]
        );

    return {
        visibleRows,
        expandedIds,
        allWbsIds,

        expansionLevel,
        maxWbsLevel,

        toggleRow,
        expandAll,
        collapseAll,
        expandToLevel
    };
}

export default useGanttTree;

import type { GanttRow as ApiGanttRow } from "../../../api/ganttApi";

export type GanttSummaryRow = ApiGanttRow & {
    weightFactor?: number | null;
    weightValue?: number | null;
    weightedProgress?: number | null;
    plannedValue?: number | null;
    actualProgress?: number | null;
    budget?: number | null;
    actualCost?: number | null;
    earnedValue?: number | null;
    cpi?: number | null;
    spi?: number | null;
};

const PROJECT_ROW_ID = "__project_level_0__";
const DAY_MS = 86_400_000;
const EPOCH_DATE_THRESHOLD = 100_000_000_000;

function finite(value: unknown, fallback = 0): number {
    const numeric = Number(value);
    return Number.isFinite(numeric) ? numeric : fallback;
}

function finiteOrNull(value: unknown): number | null {
    const numeric = Number(value);
    return Number.isFinite(numeric) ? numeric : null;
}

function clamp(value: number, min: number, max: number): number {
    return Math.max(min, Math.min(max, value));
}

/**
 * Only epoch-like numeric values are valid calendar timestamps here.
 * Relative CPM values such as 0, 1, 2 must never be interpreted as dates.
 */
function parseScheduleDate(value: unknown): Date | null {
    if (value instanceof Date) {
        const time = value.getTime();
        return Number.isFinite(time) ? new Date(time) : null;
    }

    if (typeof value === "number") {
        if (!Number.isFinite(value) || value < EPOCH_DATE_THRESHOLD) return null;
        const date = new Date(value);
        return Number.isNaN(date.getTime()) ? null : date;
    }

    if (typeof value !== "string" || value.trim().length === 0) return null;

    const trimmed = value.trim();
    const numeric = Number(trimmed);
    if (Number.isFinite(numeric)) {
        if (numeric < EPOCH_DATE_THRESHOLD) return null;
        const date = new Date(numeric);
        return Number.isNaN(date.getTime()) ? null : date;
    }

    const date = new Date(trimmed);
    return Number.isNaN(date.getTime()) ? null : date;
}

function toLocalMidnight(date: Date | null): Date | null {
    if (!date) return null;
    return new Date(date.getFullYear(), date.getMonth(), date.getDate());
}

function toLocalMidnightIso(date: Date | null): string | null {
    const normalized = toLocalMidnight(date);
    if (!normalized) return null;

    const year = normalized.getFullYear();
    const month = String(normalized.getMonth() + 1).padStart(2, "0");
    const day = String(normalized.getDate()).padStart(2, "0");

    return `${year}-${month}-${day}T00:00:00`;
}

/**
 * PCS display duration rule:
 * Finish - Start + 1 calendar day.
 *
 * This is deliberately separate from working-duration values used by the
 * scheduler/CPM engine.
 */
function inclusiveDuration(start: Date | null, finish: Date | null): number {
    const normalizedStart = toLocalMidnight(start);
    const normalizedFinish = toLocalMidnight(finish);

    if (!normalizedStart || !normalizedFinish) return 0;

    return Math.max(
        1,
        Math.floor((normalizedFinish.getTime() - normalizedStart.getTime()) / DAY_MS) + 1
    );
}

function getStart(row: ApiGanttRow): Date | null {
    return parseScheduleDate(row.start)
        ?? parseScheduleDate(row.plannedStart)
        ?? parseScheduleDate(row.actualStart);
}

function getFinish(row: ApiGanttRow): Date | null {
    return parseScheduleDate(row.finish)
        ?? parseScheduleDate(row.plannedFinish)
        ?? parseScheduleDate(row.actualFinish);
}

function flattenActivities(
    rows: ApiGanttRow[],
    output: ApiGanttRow[] = []
): ApiGanttRow[] {
    for (const row of rows) {
        if (row.type === "activity") {
            output.push(row);
        }

        if (row.children?.length) {
            flattenActivities(row.children, output);
        }
    }

    return output;
}

function flattenAll(
    rows: ApiGanttRow[],
    output: ApiGanttRow[] = []
): ApiGanttRow[] {
    for (const row of rows) {
        output.push(row);

        if (row.children?.length) {
            flattenAll(row.children, output);
        }
    }

    return output;
}

function aggregateRows(rows: ApiGanttRow[]) {
    const activities = flattenActivities(rows);
    const sourceRows = activities.length > 0 ? activities : flattenAll(rows);

    const starts = sourceRows
        .map(getStart)
        .filter((value): value is Date => value !== null);

    const finishes = sourceRows
        .map(getFinish)
        .filter((value): value is Date => value !== null);

    const start = starts.length > 0
        ? new Date(Math.min(...starts.map(value => value.getTime())))
        : null;

    const finish = finishes.length > 0
        ? new Date(Math.max(...finishes.map(value => value.getTime())))
        : null;

    const budget = sourceRows.reduce(
        (sum, row) => sum + finite((row as GanttSummaryRow).budget),
        0
    );

    const actualCost = sourceRows.reduce(
        (sum, row) => sum + finite((row as GanttSummaryRow).actualCost),
        0
    );

    const earnedValue = sourceRows.reduce(
        (sum, row) => sum + finite((row as GanttSummaryRow).earnedValue),
        0
    );

    const plannedValue = sourceRows.reduce(
        (sum, row) => sum + finite((row as GanttSummaryRow).plannedValue),
        0
    );

    const progress = budget > 0
        ? clamp((earnedValue / budget) * 100, 0, 100)
        : activities.length > 0
            ? clamp(
                activities.reduce(
                    (sum, row) => sum + finite((row as GanttSummaryRow).progress),
                    0
                ) / activities.length,
                0,
                100
            )
            : 0;

    const duration = inclusiveDuration(start, finish);

    const totalFloatValues = sourceRows
        .map(row => finiteOrNull(row.totalFloat ?? row.float))
        .filter((value): value is number => value !== null);

    const freeFloatValues = sourceRows
        .map(row => finiteOrNull(row.freeFloat))
        .filter((value): value is number => value !== null);

    const drivingFloatValues = sourceRows
        .map(row => finiteOrNull(row.drivingFloat))
        .filter((value): value is number => value !== null);

    const totalFloat = totalFloatValues.length > 0
        ? Math.min(...totalFloatValues)
        : 0;

    const freeFloat = freeFloatValues.length > 0
        ? Math.min(...freeFloatValues)
        : 0;

    const drivingFloat = drivingFloatValues.length > 0
        ? Math.min(...drivingFloatValues)
        : 0;

    const critical =
        sourceRows.some(row => row.isCritical) ||
        (totalFloat <= 0 && sourceRows.length > 0);

    const milestones = sourceRows.filter(row => row.isMilestone).length;

    const completed =
        sourceRows.length > 0 &&
        sourceRows.every(row => finite((row as GanttSummaryRow).progress) >= 100);

    const cpi = actualCost > 0 ? earnedValue / actualCost : null;
    const spi = plannedValue > 0 ? earnedValue / plannedValue : null;

    return {
        start,
        finish,
        duration,
        budget,
        actualCost,
        earnedValue,
        plannedValue,
        progress,
        totalFloat,
        freeFloat,
        drivingFloat,
        critical,
        milestones,
        completed,
        cpi,
        spi,
        sourceCount: sourceRows.length
    };
}

function buildWbsSummary(
    row: ApiGanttRow,
    parentBudget: number | null
): GanttSummaryRow {
    const sourceRow = row as GanttSummaryRow;
    const existingBudget = finiteOrNull(sourceRow.budget);

    const children = (row.children ?? []).map(child =>
        child.type === "wbs"
            ? buildWbsSummary(child, existingBudget)
            : { ...child }
    );

    const summary = aggregateRows(children.length > 0 ? children : [row]);

    const budget = existingBudget !== null && existingBudget > 0
        ? existingBudget
        : summary.budget;

    const weightFactor = parentBudget !== null && parentBudget > 0
        ? clamp((budget / parentBudget) * 100, 0, 100)
        : finiteOrNull(sourceRow.weightFactor) ?? 100;

    const weightValue = finiteOrNull(sourceRow.weightValue) ?? budget;

    const rolledStart = summary.start;
    const rolledFinish = summary.finish;
    const rolledDuration = inclusiveDuration(rolledStart, rolledFinish);

    const calculatedProgress = summary.progress;

    return {
        ...row,
        level: Math.max(1, finite(row.level)),

        // P6-style WBS roll-up: summary dates come exclusively from descendants.
        // The API contract requires numeric Start/Finish, therefore 0 means no date.
        start: rolledStart?.getTime() ?? 0,
        finish: rolledFinish?.getTime() ?? 0,
        duration: rolledDuration,

        plannedStart:
            toLocalMidnightIso(rolledStart) ??
            row.plannedStart ??
            null,

        plannedFinish:
            toLocalMidnightIso(rolledFinish) ??
            row.plannedFinish ??
            null,

        remainingDuration: Math.max(
            0,
            rolledDuration * (1 - calculatedProgress / 100)
        ),

        progress: calculatedProgress,
        actualProgress: calculatedProgress,
        weightedProgress: calculatedProgress,

        budget,
        actualCost: summary.actualCost > 0
            ? summary.actualCost
            : finite(sourceRow.actualCost),
        earnedValue: summary.earnedValue > 0
            ? summary.earnedValue
            : finite(sourceRow.earnedValue),
        plannedValue: summary.plannedValue,

        totalFloat: summary.totalFloat,
        freeFloat: summary.freeFloat,
        drivingFloat: summary.drivingFloat,
        float: summary.totalFloat,

        isCritical: summary.critical,
        status: summary.completed
            ? "Completed"
            : calculatedProgress > 0
                ? "InProgress"
                : row.status ?? "NotStarted",

        weightFactor,
        weightValue,
        cpi: summary.cpi,
        spi: summary.spi,

        children
    };
}

export function buildGanttSummaryRows(
    rows: ApiGanttRow[],
    projectDuration?: number | null,
    projectName?: string | null
): ApiGanttRow[] {
    // projectDuration is intentionally not used to manufacture a project span.
    // The Project row must be calculated from its descendant activity dates.
    void projectDuration;

    const sourceRows = rows.map(row =>
        row.type === "wbs"
            ? buildWbsSummary(row, null)
            : { ...row }
    );

    const aggregate = aggregateRows(sourceRows);
    const projectProgress = aggregate.progress;
    const projectDurationDays = inclusiveDuration(aggregate.start, aggregate.finish);

    const projectRow: GanttSummaryRow = {
        id: PROJECT_ROW_ID,
        type: "wbs",
        parentId: null,
        code: "0",
        name: projectName?.trim() || "Project Summary",
        level: 0,

        // Project Start/Finish are the min/max of all descendant activities.
        start: aggregate.start?.getTime() ?? 0,
        finish: aggregate.finish?.getTime() ?? 0,
        duration: projectDurationDays,

        plannedStart: toLocalMidnightIso(aggregate.start),
        plannedFinish: toLocalMidnightIso(aggregate.finish),
        actualStart: null,
        actualFinish: null,

        remainingDuration: Math.max(
            0,
            projectDurationDays * (1 - projectProgress / 100)
        ),

        constraintType: "ASAP",
        constraintDate: null,

        progress: projectProgress,
        actualProgress: projectProgress,
        weightedProgress: projectProgress,

        budget: aggregate.budget,
        actualCost: aggregate.actualCost,
        earnedValue: aggregate.earnedValue,
        plannedValue: aggregate.plannedValue,

        totalFloat: aggregate.totalFloat,
        freeFloat: aggregate.freeFloat,
        drivingFloat: aggregate.drivingFloat,
        float: aggregate.totalFloat,

        isCritical: aggregate.critical,
        isMilestone: false,

        status: aggregate.completed
            ? "Completed"
            : projectProgress > 0
                ? "InProgress"
                : "NotStarted",

        weightFactor: 100,
        weightValue: 100,

        cpi: aggregate.cpi,
        spi: aggregate.spi,

        predecessors: [],
        successors: [],
        children: sourceRows
    };

    return [projectRow];
}

import {
  AuthContext,

function createEmptyWorkspace(): Workspace {
  return {
    company: {
      id: "",
      code: "",
      name: "",

      id: "",

function loadStoredWorkspace():
  Workspace | null {
  try {
    const stored =
      localStorage.getItem(
        STORAGE_KEY
      );

    if (!stored) {
      return null;
    }

    const parsed =
      JSON.parse(

    if (

      !parsed.plant ||
      !parsed.project
    ) {
      return null;

    return {
      company:
        parsed.company,
      plant:
        parsed.plant,
      project:
        parsed.project,

      "Failed to restore PCS workspace",

    );

  }

export function WorkspaceProvider({
  children,
}: Props) {
  const auth =
    useContext(AuthContext);

  const isAuthenticated =
    auth?.isAuthenticated ?? false;

  const authLoading =
    auth?.loading ?? false;

  const [
    workspace,
    setWorkspaceState,
  ] = useState<Workspace>(
    createEmptyWorkspace()
  );

  const [
    companies,
    setCompanies,
  ] = useState<WorkspaceCompany[]>([]);

  const [
    plants,
    setPlants,
  ] = useState<WorkspacePlant[]>([]);

  const [
    projects,
    setProjects,
  ] = useState<WorkspaceProject[]>([]);

  useEffect(() => {
    let cancelled = false;

    async function loadWorkspace() {
      if (authLoading) {
        return;
      }

      if (!isAuthenticated) {
        if (cancelled) {
          return;
        }

        setWorkspaceState(
          createEmptyWorkspace()
        );

        setCompanies([]);
        setPlants([]);
        setProjects([]);

        clearStoredWorkspace();

        return;
      }

      try {
        const [
          companyData,
          plantData,
          projectData,
        ] = await Promise.all([
          CompanyService.getAll(),
          PlantService.getAll(),
          ProjectService.getAll(),
        ]);

        if (cancelled) {
          return;
        }

        const realCompanies =
          companyData.map(
            toWorkspaceCompany
          );

        const realPlants =
          plantData.map(
            toWorkspacePlant
          );

        const realProjects =
          projectData.map(
            toWorkspaceProject
          );

        setCompanies(
          realCompanies
        );

        setPlants(
          realPlants
        );

        setProjects(
          realProjects
        );

        if (
          realCompanies.length === 0
        ) {
          setWorkspaceState(
            createEmptyWorkspace()
          );

          clearStoredWorkspace();

          return;
        }

        const stored =
          loadStoredWorkspace();

        const selectedCompany =
          realCompanies.find(
            company =>
              company.id ===
              stored?.company.id
          ) ??
          realCompanies[0];

        const companyPlants =
          realPlants.filter(
            plant =>
              plant.companyId ===
              selectedCompany.id
          );

        const selectedPlant =
          companyPlants.find(
            plant =>
              plant.id ===
              stored?.plant.id
          ) ??
          companyPlants[0];

        const plantProjects =
          selectedPlant
            ? realProjects.filter(
                project =>
                  project.plantId ===
                  selectedPlant.id
              )
            : [];

        const selectedProject =
          plantProjects.find(
            project =>
              project.id ===
              stored?.project.id
          ) ??
          plantProjects[0];

        const nextWorkspace:
          Workspace = {
          company:
            selectedCompany,

          plant:
            selectedPlant ?? {
              id: "",
              companyId:
                selectedCompany.id,
              code: "",
              name: "",
            },

          project:
            selectedProject ?? {
              id: "",
              plantId:
                selectedPlant?.id ?? "",
              code: "",
              name: "",
            },
        };

        setWorkspaceState(
          nextWorkspace
        );

        try {
          localStorage.setItem(
            STORAGE_KEY,
            JSON.stringify(
              nextWorkspace
            )
          );
        }
        catch (error) {
          console.warn(
            "Failed to persist PCS workspace",
            error
          );
        }
      }
      catch (error) {
        if (cancelled) {
          return;
        }

        console.error(
          "Failed to load PCS workspace",
          error
        );

        const stored =
          loadStoredWorkspace();

        if (stored) {
          setWorkspaceState(
            stored
          );
        }
        else {
          setWorkspaceState(
            createEmptyWorkspace()
          );
        }
      }
    }

    loadWorkspace();

    return () => {
      cancelled = true;
    };
  }, [
    isAuthenticated,
    authLoading,
  ]);

  function persistWorkspace(
    nextWorkspace: Workspace
  ): void {
    setWorkspaceState(
      nextWorkspace
    );

    try {
      localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(
          nextWorkspace
        )
      );
    }
    catch (error) {
      console.warn(
        "Failed to persist PCS workspace",
        error
      );
    }
  }

  function setCompany(
    company: WorkspaceCompany
  ): void {
    const companyPlants =
      plants.filter(
        plant =>
          plant.companyId ===
          company.id
      );

    const nextPlant =
      companyPlants[0] ?? {
        id: "",
        companyId:
          company.id,
        code: "",
        name: "",
      };

    const plantProjects =
      projects.filter(
        project =>
          project.plantId ===
          nextPlant.id
      );

    const nextProject =
      plantProjects[0] ?? {
        id: "",
        plantId:
          nextPlant.id,
        code: "",
        name: "",
      };

    persistWorkspace({
      company,
      plant:
        nextPlant,
      project:
        nextProject,
    });
  }

  function setPlant(
    plant: WorkspacePlant
  ): void {
    const plantProjects =
      projects.filter(
        project =>
          project.plantId ===
          plant.id
      );

    const nextProject =
      plantProjects[0] ?? {
        id: "",
        plantId:
          plant.id,
        code: "",
        name: "",
      };

    persistWorkspace({
      ...workspace,
      plant,
      project:
        nextProject,
    });
  }

  function setProject(
    project: WorkspaceProject
  ): void {
    persistWorkspace({
      ...workspace,
      project,
    });
  }

  const value =
    useMemo(
      () => ({
        workspace,
        companies,
        plants,
        projects,
        setCompany,
        setPlant,
        setProject,
      }),
      [
        workspace,
        companies,
        plants,
        projects,
      ]
    );

  return (
    <WorkspaceContext.Provider
      value={value}
    >
      {children}
    </WorkspaceContext.Provider>
  );
}
